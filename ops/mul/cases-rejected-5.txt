def run_ort_batch(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """One Mul node over two arrays of the same type, in a single runtime call."""
    proto = helper.np_dtype_to_tensor_dtype(a.dtype)
    vi_a = helper.make_tensor_value_info("A", proto, list(a.shape))
    vi_b = helper.make_tensor_value_info("B", proto, list(b.shape))
    vi_c = helper.make_tensor_value_info("C", proto, None)
    node = helper.make_node(OP, ["A", "B"], ["C"], name="mul0")
    graph = helper.make_graph([node], "mul_graph", [vi_a, vi_b], [vi_c])
    model = helper.make_model(graph, opset_imports=[helper.make_opsetid("", OPSET)],
                              ir_version=IR_VERSION)
    sess = ort.InferenceSession(model.SerializeToString(),
                                providers=["CPUExecutionProvider"])
    (c,) = sess.run(["C"], {"A": a, "B": b})
    return c


def _bits_equal(want, got) -> bool:
    """Bit-for-bit equality, NaN equal to NaN, of two arrays of the same type.

    The sign of a zero is part of the bit pattern, so +0.0 and -0.0 are different
    results here: the document specifies the sign of a null product.
    """
    want = np.asarray(want)
    got = np.asarray(got)
    if want.shape != got.shape or want.dtype != got.dtype:
        return False
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        return bool(np.all((bw == bg) | both_nan))
    return bool(np.all(want == got))


def _diff_indices(want, got, limit=5):
    """Flat indices where two arrays differ (NaN equal to NaN); None on shape/type."""
    want = np.asarray(want)
    got = np.asarray(got)
    if want.shape != got.shape or want.dtype != got.dtype:
        return None
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        mask = ~((bw == bg) | both_nan)
    else:
        mask = ~(want == got)
    idx = [int(i) for i in np.nonzero(mask.ravel())[0]]
    return idx if limit is None else idx[:limit]


# ---------------------------------------------------------------------------
# exact arithmetic: the reference the document itself states
# ---------------------------------------------------------------------------
#
# The float formula is round(A * B): the exact product in the reals, rounded to the
# nearest value of the type with roundTiesToEven and an unbounded exponent range.  It is
# computed here from the operands' exact values (Fraction of the float, never of the
# decimal literal used to build the array), independently of the platform's arithmetic.


def _pow2(e: int) -> Fraction:
    return Fraction(2) ** e


def _round_half_even(x: Fraction) -> int:
    """roundTiesToEven of a non-negative rational to the nearest integer."""
    q = x.numerator // x.denominator
    r = x - q
    half = Fraction(1, 2)
    if r > half:
        return q + 1
    if r < half:
        return q
    return q if q % 2 == 0 else q + 1


def _frac_round_to_type(value: Fraction, dtype):
    """Round an exact rational to the nearest value of `dtype`, ties to even.

    The exponent range is unbounded, as the document states: a magnitude whose nearest
    value exceeds the largest finite value of the type gives an infinity of its sign.
    """
    dt = np.dtype(dtype)
    f = np.finfo(dt)
    p = f.nmant + 1                       # significand bits, implicit bit included
    emin = f.minexp                       # exponent of the smallest normal
    max_finite = Fraction(float(f.max))
    if value == 0:
        return dt.type(0.0)
    sign = -1.0 if value < 0 else 1.0
    a = abs(value)
    e = a.numerator.bit_length() - a.denominator.bit_length()
    while _pow2(e) > a:
        e -= 1
    while _pow2(e + 1) <= a:
        e += 1
    if e < emin:
        e = emin                          # the subnormal range: uniform spacing
    spacing = _pow2(e - p + 1)
    n = _round_half_even(a / spacing)
    result = n * spacing
    if result > max_finite:
        return dt.type(sign * float("inf"))
    return dt.type(sign * float(result))


def _exact_mul_value(x, y, dtype):
    """C[i] of the float formula: NaN, a signed infinity, or round(x * y)."""
    dt = np.dtype(dtype)
    fx = float(x)
    fy = float(y)
    sx = bool(np.signbit(fx))
    sy = bool(np.signbit(fy))
    if np.isnan(fx) or np.isnan(fy):
        return dt.type(float("nan"))
    if np.isinf(fx) or np.isinf(fy):
        if fx == 0.0 or fy == 0.0:
            return dt.type(float("nan"))       # the invalid operation of IEEE 754 7.2
        return dt.type(float("-inf") if (sx != sy) else float("inf"))
    prod = Fraction(fx) * Fraction(fy)
    if prod == 0:
        return dt.type(-0.0 if (sx != sy) else 0.0)
    return _frac_round_to_type(prod, dt)


def _double_rounded_float16(x, y):
    """The same product rounded through float32 first, or None for the wider types.

    A kernel that computes a float16 product in float32 and narrows the result rounds
    twice; on the rare operand pair whose exact product sits a hair from a float16
    midpoint that differs from the single rounding the document states.  Recorded, not
    required.
    """
    if np.dtype(np.float16) is None:
        return None
    fx = float(x)
    fy = float(y)
    if np.isnan(fx) or np.isnan(fy) or np.isinf(fx) or np.isinf(fy):
        return None
    if fx == 0.0 or fy == 0.0:
        return None
    prod = Fraction(fx) * Fraction(fy)
    f32 = _frac_round_to_type(prod, np.float32)
    with np.errstate(all="ignore"):
        return np.float16(float(f32))

#
# A second, literal reading of **Mul** (int, int) and **Mul** (uint, uint), in exact
# arithmetic: C[i] is the value of the n-bit type congruent to the exact product modulo
# 2^n.  A disagreement here is a defect of the document, not of the implementation.


def _doc_int_array(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    dt = a.dtype
    n = dt.itemsize * 8
    mod = 1 << n
    xs = a.ravel().tolist()
    ys = b.ravel().tolist()
    if dt.kind == "i":
        lo = -(1 << (n - 1))
        hi = (1 << (n - 1)) - 1
        out = []
        for x, y in zip(xs, ys):
            u = (int(x) * int(y)) % mod
            out.append(u if u <= hi else u - mod)
        return np.array(out, dtype=dt).reshape(a.shape)
    out = [(int(x) * int(y)) % mod for x, y in zip(xs, ys)]
    return np.array(out, dtype=dt).reshape(a.shape)


def _rand_values(rng, dt, n):
    """n random values of `dt`, drawn uniformly over the whole bit patterns of the type."""
    hi = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    lo = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    raw = (hi << np.uint64(32)) | lo
    bits = dt.itemsize * 8
    if bits < 64:
        raw = raw & np.uint64((1 << bits) - 1)
    return raw.astype(np.dtype("u%d" % dt.itemsize)).view(dt)

def sweep_schema_types(impl):
    """The document's declared type families against the ONNX schema's admitted set.

    A type the document declares that the schema does not admit makes every case of that
    type open at model resolution, and a case set that drops it would converge on the
    wrong document.
    """
    declared = ["tensor(float16)", "tensor(float)", "tensor(double)",
                "tensor(int8)", "tensor(int16)", "tensor(int32)", "tensor(int64)",
                "tensor(uint8)", "tensor(uint16)", "tensor(uint32)", "tensor(uint64)"]
    try:
        from onnx import defs
        schema = defs.get_schema(OP, OPSET)
        allowed = set()
        for constraint in schema.type_constraints:
            if constraint.type_param_str == "T":
                allowed |= set(constraint.allowed_type_strs)
    except Exception as exc:
        return {"sweep": "the document's types against the ONNX schema",
                "summary": "the schema of %s-%d could not be read (%s: %s)"
                           % (OP, OPSET, type(exc).__name__, exc),
                "failures": []}
    failures = []
    for name in declared:
        if name not in allowed:
            failures.append({"who": "the document declares %s" % name,
                             "detail": "the schema of %s-%d admits %s for T"
                                       % (OP, OPSET, sorted(allowed))})
    summary = ("%d declared types, %s admitted by the schema of %s-%d"
               % (len(declared),
                  "every one" if not failures else "not every one", OP, OPSET))
    extra = sorted(allowed - set(declared))
    if extra:
        summary += "; the schema also admits %s, which the document does not specify" \
                   % ", ".join(extra)
    return {"sweep": "the document's types against the ONNX schema",
            "summary": summary, "failures": failures}


def sweep_spec_examples(impl):
    """Every worked example of the typed sections: as documented, vs ORT, vs the impl."""
    failures = []
    examples = _spec_examples()
    for label, a, b, documented in examples:
        try:
            ref = run_ort_batch(a, b)
        except Exception as exc:
            ref = None
            failures.append({"who": label,
                             "detail": "the runtime refused the model: %s: %s"
                                       % (type(exc).__name__, exc)})
        if ref is not None and not _bits_equal(documented, ref):
            failures.append({"who": label, "detail": "documented %s, ORT %s"
                             % (documented.tolist(), np.asarray(ref).tolist())})
        try:
            got = np.asarray(impl.mul(a, b))
        except Exception as exc:
            failures.append({"who": label,
                             "detail": "the implementation raised %s: %s"
                                       % (type(exc).__name__, exc)})
            continue
        if not _bits_equal(documented, got):
            failures.append({"who": label,
                             "detail": "documented %s, implementation %s"
                                       % (documented.tolist(), got.tolist())})
    return {"sweep": "the document's worked examples",
            "summary": "%d examples, as documented vs ORT vs the implementation"
                       % len(examples),
            "failures": failures}


def sweep_special_values(impl):
    """The special values of the float types, decided by the document's own rule.

    Infinity and NaN are not real numbers: they are decided by the case analysis of the
    float formula before any exact arithmetic is reached, and `_exact_mul_value` does
    exactly that.  The sweep checks the document's statements against that rule and the
    implementation against both.  ONNX Runtime's value is recorded, not required, where
    it departs: a kernel is free to be less accurate than the document.
    """
    failures = []
    rows = _special_statements()
    recorded = 0
    for label, x, y, dt, expected in rows:
        a = np.array([x], dtype=dt)
        b = np.array([y], dtype=dt)
        exact = _exact_mul_value(a[0], b[0], dt)
        exact_arr = np.array([exact], dtype=dt)
        if expected is None:
            if not bool(np.isnan(exact)):
                failures.append({"who": label,
                                 "detail": "the document's rule gives %r where the "
                                           "statement says NaN" % float(exact)})
        else:
            want = np.array([expected], dtype=dt)
            if not _bits_equal(want, exact_arr):
                failures.append({"who": label,
                                 "detail": "the document's rule gives %r where the "
                                           "statement says %r"
                                           % (float(exact), float(want[0]))})
        try:
            ref = run_ort_batch(a, b)
        except Exception:
            ref = None
        if ref is not None and not _bits_equal(exact_arr, np.asarray(ref)):
            recorded += 1
        try:
            got = np.asarray(impl.mul(a, b))
        except Exception as exc:
            failures.append({"who": label,
                             "detail": "the implementation raised %s: %s"
                                       % (type(exc).__name__, exc)})
            continue
        if not _bits_equal(exact_arr, got):
            failures.append({"who": label,
                             "detail": "document %r, implementation %r"
                                       % (float(exact),
                                          float(np.asarray(got).ravel()[0]))})
    summary = ("%d statements of the float section checked against the document's own "
               "case analysis; %d runtime values recorded, not required"
               % (len(rows), recorded))
    return {"sweep": "the special values of the float types (document's rule)",
            "summary": summary, "failures": failures}


def sweep_int_domain(impl):
    """The document's integer case analysis against ONNX Runtime and the implementation.

    The two 8-bit types are covered exhaustively (all 65536 pairs each); the wider types
    use their boundary values and 200 random pairs.  The literal transcription of the
    document is compared with the runtime as well: a disagreement there is a defect of
    the document.
    """
    rng = np.random.default_rng(20261003)
    rows, failures = [], []
    for dtype in ALL_INT:
        dt = np.dtype(dtype)
        info = np.iinfo(dt)
        exhaustive = dt.itemsize == 1
        if exhaustive:
            xs = np.arange(int(info.min), int(info.max) + 1, dtype=np.int64)
            aa = np.repeat(xs, len(xs)).astype(dt)
            bb = np.tile(xs, len(xs)).astype(dt)
        else:
            if dt.kind == "u":
                mx = int(info.max)
                edges = [0, 1, 2, mx // 2, mx // 2 + 1, mx - 1, mx]
            else:
                mn, mx = int(info.min), int(info.max)
                edges = [mn, mn + 1, -1, 0, 1, 2, mx - 1, mx]
            pairs = [(x, y) for x in edges for y in edges]
            ra = _rand_values(rng, dt, 200)
            rb = _rand_values(rng, dt, 200)
            pairs += list(zip(ra.tolist(), rb.tolist()))
            aa = np.array([p[0] for p in pairs], dtype=dt)
            bb = np.array([p[1] for p in pairs], dtype=dt)

        want = _doc_int_array(aa, bb)
        try:
            ref = np.asarray(run_ort_batch(aa, bb))
        except Exception as exc:
            ref = None
            failures.append({"type": dt.name, "who": "ONNX Runtime",
                             "detail": "the runtime refused the model: %s: %s"
                                       % (type(exc).__name__, exc)})
        try:
            got = np.asarray(impl.mul(aa, bb))
        except Exception as exc:
            failures.append({"type": dt.name, "who": "the implementation",
                             "detail": "raised %s: %s"
                                       % (type(exc).__name__, exc)})
            continue

        if ref is not None:
            bad = _diff_indices(want, ref, limit=5)
            if bad is None:
                failures.append({"type": dt.name, "who": "the document vs ONNX Runtime",
                                 "detail": "shape %s or type %s differs from the "
                                           "document's %s / %s"
                                           % (ref.shape, ref.dtype,
                                              want.shape, want.dtype)})
            elif bad:
                failures.append({
                    "type": dt.name, "who": "the document vs ONNX Runtime",
                    "detail": "; ".join("doc: %d * %d = %d, ort: %d"
                                        % (int(aa.ravel()[i]), int(bb.ravel()[i]),
                                           int(want.ravel()[i]), int(ref.ravel()[i]))
                                        for i in bad)})

        bad = _diff_indices(want, got, limit=5)
        if bad is None:
            failures.append({"type": dt.name,
                             "who": "the implementation vs the document",
                             "detail": "shape %s or type %s differs from the "
                                       "document's %s / %s"
                                       % (got.shape, got.dtype,
                                          want.shape, want.dtype)})
        elif bad:
            failures.append({
                "type": dt.name, "who": "the implementation vs the document",
                "detail": "; ".join("%d * %d: doc %d, impl %d"
                                    % (int(aa.ravel()[i]), int(bb.ravel()[i]),
                                       int(want.ravel()[i]), int(got.ravel()[i]))
                                    for i in bad)})
        rows.append("%s %s" % (dt.name,
                               "exhaustive (65536 pairs)" if exhaustive
                               else "%d pairs" % len(aa)))
    return {"sweep": "int domain (document vs ORT vs implementation)",
            "summary": "%d types: %s" % (len(rows), ", ".join(rows)),
            "failures": failures}
