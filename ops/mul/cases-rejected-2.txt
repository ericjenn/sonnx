OP = "Mul"

NODE_OUTPUTS = ("out",)

ALL_FLOAT = [np.float16, np.float32, np.float64]
SIGNED_INT = [np.int8, np.int16, np.int32, np.int64]
UNSIGNED_INT = [np.uint8, np.uint16, np.uint32, np.uint64]

UINT_OF_WIDTH = {2: np.uint16, 4: np.uint32, 8: np.uint64}

def run_ort_batch(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """One Mul node over two arrays of the same type, in a single runtime call."""
    proto = helper.np_dtype_to_tensor_dtype(a.dtype)
    vi_a = helper.make_tensor_value_info("A", proto, list(a.shape))
    vi_b = helper.make_tensor_value_info("B", proto, list(b.shape))
    vi_c = helper.make_tensor_value_info("C", proto, None)
    node = helper.make_node(OP, ["A", "B"], ["C"], name="mul0")
    graph = helper.make_graph([node], "mul_graph", [vi_a, vi_b], [vi_c])
    model = helper.make_model(graph, opset_imports=[helper.make_opsetid("", OPSET)],
                              ir_version=IR_VERSION)
    sess = ort.InferenceSession(model.SerializeToString(),
                                providers=["CPUExecutionProvider"])
    (c,) = sess.run(["C"], {"A": a, "B": b})
    return c


def _bits_equal(want: np.ndarray, got: np.ndarray) -> bool:
    """Bit-for-bit equality, NaN equal to NaN, of two arrays of the same type.

    The sign of a zero and the sign of an infinity are part of the bit pattern, so
    +0.0 and -0.0 are different results here: the document specifies the sign of a
    null result and of an infinite one.
    """
    if want.shape != got.shape or want.dtype != got.dtype:
        return False
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        return bool(np.all((bw == bg) | both_nan))
    return bool(np.all(want == got))


def _diff_indices(want: np.ndarray, got: np.ndarray, limit=5):
    """Flat indices where two arrays differ (NaN equal to NaN); None on shape/type."""
    if want.shape != got.shape or want.dtype != got.dtype:
        return None
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        mask = ~((bw == bg) | both_nan)
    else:
        mask = ~(want == got)
    idx = [int(i) for i in np.nonzero(mask.ravel())[0]]
    return idx if limit is None else idx[:limit]


def _same_bits(want, got) -> bool:
    return _bits_equal(np.asarray(want).ravel(), np.asarray(got).ravel())


# ---------------------------------------------------------------------------
# exact arithmetic: the reference the document itself states
# ---------------------------------------------------------------------------
#
# The float formula is: NaN for a NaN or for a zero times an infinity; a signed
# infinity for an infinity times a nonzero non-NaN; and otherwise round(A * B), the
# exact real product rounded to the nearest value of the type with roundTiesToEven and
# an unbounded exponent range.  That is computed here from the operands' exact values
# (Fraction of the float, never of the decimal literal), independently of the
# platform's library and of the runtime.


def _pow2(e: int) -> Fraction:
    return Fraction(2) ** e


def _round_half_even(x: Fraction) -> int:
    """roundTiesToEven of a non-negative rational to the nearest integer."""
    q = x.numerator // x.denominator
    r = x - q
    half = Fraction(1, 2)
    if r > half:
        return q + 1
    if r < half:
        return q
    return q if q % 2 == 0 else q + 1


def _frac_round_to_type(value: Fraction, dtype):
    """Round an exact rational to the nearest value of `dtype`, ties to even.

    The exponent range is unbounded, as the document states: a magnitude whose nearest
    value exceeds the largest finite value of the type gives an infinity of its sign,
    and a magnitude below half the smallest subnormal gives a zero of its sign.
    """
    dt = np.dtype(dtype)
    f = np.finfo(dt)
    p = f.nmant + 1                       # significand bits, implicit bit included
    emin = f.minexp                       # exponent of the smallest normal
    max_finite = Fraction(float(f.max))
    if value == 0:
        return dt.type(0.0)
    sign = -1.0 if value < 0 else 1.0
    a = abs(value)
    e = a.numerator.bit_length() - a.denominator.bit_length()
    while _pow2(e) > a:
        e -= 1
    while _pow2(e + 1) <= a:
        e += 1
    if e < emin:
        e = emin                          # the subnormal range: uniform spacing
    spacing = _pow2(e - p + 1)
    n = _round_half_even(a / spacing)
    result = n * spacing
    if result > max_finite:
        return dt.type(sign * float("inf"))
    if result == 0:
        return dt.type(sign * 0.0)
    return dt.type(sign * float(result))


def _exact_float_result(x, y, dtype):
    """C[i] of **Mul** (float, float), the document's three cases, in exact arithmetic."""
    dt = np.dtype(dtype)
    fx, fy = float(x), float(y)
    if np.isnan(fx) or np.isnan(fy):
        return dt.type(float("nan"))
    if np.isinf(fx) or np.isinf(fy):
        if fx == 0.0 or fy == 0.0:
            return dt.type(float("nan"))          # the invalid operation of IEEE 7.2
        sign = -1.0 if (np.signbit(fx) != np.signbit(fy)) else 1.0
        return dt.type(sign * float("inf"))
    p = Fraction(fx) * Fraction(fy)
    if p == 0:
        sign = -1.0 if (np.signbit(fx) != np.signbit(fy)) else 1.0
        return dt.type(sign * 0.0)
    return _frac_round_to_type(p, dt)


# ---------------------------------------------------------------------------
# the integer case analysis of the document, transcribed literally
# ---------------------------------------------------------------------------
#
# C[i] is the value of the signed (resp. unsigned) n-bit type congruent to the exact
# product modulo 2^n: u = product mod 2^n in [0, 2^n-1], then u - 2^n when the type is
# signed and u >= 2^(n-1).  The transcription is a second, literal reading of the two
# integer sections, independent of the implementation and of the runtime.


def _doc_int_array(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    dt = a.dtype
    n = dt.itemsize * 8
    modulus = 2 ** n
    if dt.itemsize <= 4:
        prod = a.astype(np.int64) * b.astype(np.int64)
        u = prod % modulus
        if dt.kind == "i":
            u = np.where(u >= 2 ** (n - 1), u - modulus, u)
        return u.astype(dt)
    flat = []
    for x, y in zip(a.ravel().tolist(), b.ravel().tolist()):
        u = (int(x) * int(y)) % modulus
        if dt.kind == "i" and u >= 2 ** (n - 1):
            u -= modulus
        flat.append(u)
    return np.array(flat, dtype=dt).reshape(a.shape)


def _rand_values(rng, dt, n):
    """n random values of `dt`, drawn uniformly over the whole bit patterns of the type."""
    hi = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    lo = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    raw = (hi << np.uint64(32)) | lo
    bits = dt.itemsize * 8
    if bits < 64:
        raw = raw & np.uint64((1 << bits) - 1)
    return raw.astype(np.dtype("u%d" % dt.itemsize)).view(dt)

def sweep_int_domain(impl):
    """The document's integer case analysis against ONNX Runtime and the implementation.

    The two 8-bit types are covered exhaustively (all 65536 pairs each); the wider types
    use their boundary values and a random sample, drawn over the whole bit patterns of
    the type.  A disagreement between the document and ONNX Runtime is a defect of the
    document, a disagreement between the document and the implementation a defect of the
    implementation.
    """
    rng = np.random.default_rng(20261003)
    rows, failures = [], []
    for dtype in ALL_INT:
        dt = np.dtype(dtype)
        info = np.iinfo(dt)
        exhaustive = dt.itemsize == 1
        if exhaustive:
            xs = np.arange(int(info.min), int(info.max) + 1, dtype=np.int64)
            aa = np.repeat(xs, xs.size).astype(dt)
            bb = np.tile(xs, xs.size).astype(dt)
        else:
            if dt.kind == "i":
                edges = [int(info.min), int(info.min) + 1, -3, -2, -1, 0, 1, 2, 3,
                         int(info.max) - 1, int(info.max)]
            else:
                edges = [0, 1, 2, 3, 4, int(info.max) // 2, int(info.max) - 1,
                         int(info.max)]
            pairs = [(x, y) for x in edges for y in edges]
            n_rand = 300 if dt.itemsize <= 4 else 120
            pairs += list(zip(_rand_values(rng, dt, n_rand).tolist(),
                              _rand_values(rng, dt, n_rand).tolist()))
            aa = np.array([p[0] for p in pairs], dtype=dt)
            bb = np.array([p[1] for p in pairs], dtype=dt)

        want = _doc_int_array(aa, bb)          # the document, literally
        try:
            got = np.asarray(impl.mul(aa, bb))
        except Exception as exc:
            failures.append({"type": dt.name, "who": "the implementation",
                             "detail": "raised %s: %s" % (type(exc).__name__, exc)})
            rows.append("%s: the implementation raised" % dt.name)
            continue

        bad = _diff_indices(want, got, limit=5)
        if bad is None:
            failures.append({"type": dt.name, "who": "the document vs the implementation",
                             "detail": "shape %s or type %s differs from the document's "
                                       "%s / %s" % (got.shape, got.dtype,
                                                    want.shape, want.dtype)})
        elif bad:
            failures.append({
                "type": dt.name, "who": "the document vs the implementation",
                "detail": "; ".join("doc: %d * %d = %d, impl: %d"
                                    % (int(aa.ravel()[i]), int(bb.ravel()[i]),
                                       int(want.ravel()[i]), int(got.ravel()[i]))
                                    for i in bad),
            })

        note = ""
        try:
            ref = run_ort_batch(aa, bb)
        except Exception as exc:
            note = "; ONNX Runtime refused the batch (%s: %s)" % (type(exc).__name__,
                                                                  exc)
        else:
            bad_ref = _diff_indices(want, ref, limit=5)
            if bad_ref is None:
                failures.append({"type": dt.name, "who": "the document vs ONNX Runtime",
                                 "detail": "shape %s or type %s differs from the "
                                           "document's %s / %s"
                                           % (ref.shape, ref.dtype, want.shape,
                                              want.dtype)})
            elif bad_ref:
                failures.append({
                    "type": dt.name, "who": "the document vs ONNX Runtime",
                    "detail": "; ".join("doc: %d * %d = %d, ort: %d"
                                        % (int(aa.ravel()[i]), int(bb.ravel()[i]),
                                           int(want.ravel()[i]), int(ref.ravel()[i]))
                                        for i in bad_ref),
                })
        rows.append("%s: %s%s"
                    % (dt.name,
                       "exhaustive (65536 pairs)" if exhaustive
                       else "%d pairs" % int(aa.size), note))

    return {"sweep": "int domain (document vs ONNX Runtime vs implementation)",
            "summary": "%d types: %s" % (len(rows), ", ".join(rows)),
            "failures": failures}
