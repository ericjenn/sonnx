def run_ort_batch(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """One Mul node over two arrays of the same type, in a single runtime call."""
    a = np.asarray(a)
    b = np.asarray(b)
    proto = helper.np_dtype_to_tensor_dtype(a.dtype)
    vi_a = helper.make_tensor_value_info("A", proto, list(a.shape))
    vi_b = helper.make_tensor_value_info("B", proto, list(b.shape))
    vi_c = helper.make_tensor_value_info("C", proto, None)
    node = helper.make_node(OP, ["A", "B"], ["C"], name="mul0")
    graph = helper.make_graph([node], "mul_graph", [vi_a, vi_b], [vi_c])
    model = helper.make_model(graph, opset_imports=[helper.make_opsetid("", OPSET)],
                              ir_version=IR_VERSION)
    sess = ort.InferenceSession(model.SerializeToString(),
                                providers=["CPUExecutionProvider"])
    (c,) = sess.run(["C"], {"A": a, "B": b})
    return c


def _bits_equal(want: np.ndarray, got: np.ndarray) -> bool:
    """Bit-for-bit equality, NaN equal to NaN, of two arrays of the same type."""
    want = np.asarray(want)
    got = np.asarray(got)
    if want.shape != got.shape or want.dtype != got.dtype:
        return False
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        return bool(np.all((bw == bg) | both_nan))
    return bool(np.all(want == got))


def _diff_indices(want: np.ndarray, got: np.ndarray, limit=5):
    """Flat indices where two arrays differ (NaN equal to NaN); None on shape/type."""
    want = np.asarray(want)
    got = np.asarray(got)
    if want.shape != got.shape or want.dtype != got.dtype:
        return None
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        mask = ~((bw == bg) | both_nan)
    else:
        mask = ~(want == got)
    idx = [int(i) for i in np.nonzero(mask.ravel())[0]]
    return idx if limit is None else idx[:limit]


# ---------------------------------------------------------------------------
# exact arithmetic: the reference the document itself states
# ---------------------------------------------------------------------------
#
# The float formula of **Mul** (float, float) is: NaN if either operand is NaN or one is a
# zero and the other an infinity; an infinity of the sign of the exact product if one
# operand is an infinity and the other is neither a zero nor NaN; round(x * y) otherwise,
# the exact product in the reals rounded to the type with roundTiesToEven and an unbounded
# exponent range.  It is computed here from the operands' exact values (Fraction of the
# float, never of the decimal literal), independently of the platform and the runtime.


def _pow2(e: int) -> Fraction:
    return Fraction(2) ** e


def _round_half_even(x: Fraction) -> int:
    """roundTiesToEven of a non-negative rational to the nearest integer."""
    q = x.numerator // x.denominator
    r = x - q
    half = Fraction(1, 2)
    if r > half:
        return q + 1
    if r < half:
        return q
    return q if q % 2 == 0 else q + 1


def _round_to_type(value: Fraction, dt: np.dtype):
    """Round an exact rational to the nearest value of `dt`, ties to even.

    The exponent range is unbounded, as the document states: a magnitude whose nearest
    value exceeds the largest finite value of the type gives an infinity of its sign.
    """
    f = np.finfo(dt)
    p = f.nmant + 1                       # significand bits, implicit bit included
    emin = f.minexp                       # exponent of the smallest normal
    max_finite = Fraction(float(f.max))
    if value == 0:
        return dt.type(0.0)
    sign = -1.0 if value < 0 else 1.0
    a = abs(value)
    e = a.numerator.bit_length() - a.denominator.bit_length()
    while _pow2(e) > a:
        e -= 1
    while _pow2(e + 1) <= a:
        e += 1
    if e < emin:
        e = emin                          # the subnormal range: uniform spacing
    spacing = _pow2(e - p + 1)
    n = _round_half_even(a / spacing)
    result = n * spacing
    if result > max_finite:
        return dt.type(sign * float("inf"))
    return dt.type(sign * float(result))


def _exact_float_result(x, y, dt: np.dtype):
    """C[i] of **Mul** (float, float), the document's three cases."""
    fx, fy = float(x), float(y)
    T = dt.type
    if np.isnan(fx) or np.isnan(fy):
        return T(np.nan)
    if np.isinf(fx) or np.isinf(fy):
        if (np.isinf(fx) and fy == 0.0) or (np.isinf(fy) and fx == 0.0):
            return T(np.nan)                      # the invalid operation of IEEE 7.2
        sign = -1.0 if bool(np.signbit(fx)) != bool(np.signbit(fy)) else 1.0
        return T(sign * float("inf"))
    sign = -1.0 if bool(np.signbit(fx)) != bool(np.signbit(fy)) else 1.0
    if fx == 0.0 or fy == 0.0:
        return T(sign * 0.0)                      # the rule of signs on a null product
    return _round_to_type(Fraction(fx) * Fraction(fy), dt)

# A second, literal reading of the two sections **Mul** (int, int) and **Mul** (uint, uint),

def doc_int_signed(x: int, y: int, n: int) -> int:
    """The case analysis of **Mul** (int, int), in exact arithmetic."""
    u = (x * y) % (2 ** n)
    return u if u <= 2 ** (n - 1) - 1 else u - 2 ** n


def doc_int_unsigned(x: int, y: int, n: int) -> int:
    """The case analysis of **Mul** (uint, uint), in exact arithmetic."""
    return (x * y) % (2 ** n)


def _doc_int_array(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    dt = a.dtype
    n = dt.itemsize * 8
    fn = doc_int_signed if dt.kind == "i" else doc_int_unsigned
    flat = [fn(int(x), int(y), n) for x, y in zip(a.ravel(), b.ravel())]
    return np.array(flat, dtype=dt).reshape(a.shape)


def _rand_values(rng, dt, n):
    """n random values of `dt`, drawn uniformly over the whole bit patterns of the type."""
    hi = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    lo = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    raw = (hi << np.uint64(32)) | lo
    bits = dt.itemsize * 8
    if bits < 64:
        raw = raw & np.uint64((1 << bits) - 1)
    return raw.astype(np.dtype("u%d" % dt.itemsize)).view(dt)

def _float_probe(dt, rng, n_rand=96):
    """The operands of the exact float sweep: the special values and a random sample."""
    f = np.finfo(dt)
    maxf = float(f.max)
    tiny = float(f.tiny)
    small = float(np.nextafter(dt.type(0.0), dt.type(1.0)))
    one_up = float(np.nextafter(dt.type(1.0), dt.type(2.0)))
    one_down = float(np.nextafter(dt.type(1.0), dt.type(0.0)))
    specials = [0.0, -0.0, 1.0, -1.0, 0.5, -0.5, 2.0, -2.0, 3.0, -3.0,
                tiny, -tiny, small, -small, 2.0 * small, -2.0 * small,
                maxf, -maxf, maxf / 2.0, -maxf / 2.0,
                one_up, one_down,
                float("inf"), float("-inf"), float("nan")]
    pairs = [(x, y) for x in specials for y in specials]
    lo, hi = int(f.minexp) - 2, int(f.maxexp)
    for _ in range(n_rand):
        ea = int(rng.integers(lo, hi + 1))
        eb = int(rng.integers(lo, hi + 1))
        va = float(np.ldexp(float(rng.uniform(-2.0, 2.0)), ea))
        vb = float(np.ldexp(float(rng.uniform(-2.0, 2.0)), eb))
        pairs.append((max(-maxf, min(maxf, va)), max(-maxf, min(maxf, vb))))
    a = np.array([p[0] for p in pairs], dtype=dt)
    b = np.array([p[1] for p in pairs], dtype=dt)
    return a, b


def sweep_float_exact(impl):
    """The float section's three cases against exact arithmetic, ORT and the implementation.

    Each element is decided by the document's own case analysis: NaN for a NaN operand or
    for a zero times an infinity; an infinity of the sign of the exact product for an
    infinity operand; round(x * y) with roundTiesToEven and an unbounded exponent range
    otherwise, computed with `Fraction` from the operands' exact values.  The special
    values of every type are operands and results here: the signed zeros, both infinities,
    NaN, the smallest subnormal, the largest finite value, and the rounding tie.
    """
    rng = np.random.default_rng(20261003)
    rows, failures = [], []
    checked = 0
    for dtype in ALL_FLOAT:
        dt = np.dtype(dtype)
        a, b = _float_probe(dt, rng)
        exact = np.array([_exact_float_result(x, y, dt)
                          for x, y in zip(a.tolist(), b.tolist())], dtype=dt)
        ref = run_ort_batch(a, b)
        got = np.asarray(impl.mul(a, b))
        checked += int(a.size)

        n_nan = int(np.isnan(exact).sum())
        n_inf = int(np.isinf(exact).sum())
        n_zero = int(((exact == 0) & ~np.isnan(exact)).sum())

        for who, other in (("ONNX Runtime", ref), ("the implementation", got)):
            bad = _diff_indices(exact, other, limit=5)
            if bad is None:
                failures.append({
                    "type": dt.name, "who": who,
                    "detail": "shape %s or type %s differs from the exact result's "
                              "%s / %s" % (other.shape, other.dtype,
                                           exact.shape, exact.dtype)})
                continue
            if bad:
                failures.append({
                    "type": dt.name,
