def run_ort_batch(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """One Div node over two arrays of the same type, in a single runtime call."""
    proto = helper.np_dtype_to_tensor_dtype(a.dtype)
    vi_a = helper.make_tensor_value_info("A", proto, list(a.shape))
    vi_b = helper.make_tensor_value_info("B", proto, list(b.shape))
    vi_c = helper.make_tensor_value_info("C", proto, None)
    node = helper.make_node(OP, ["A", "B"], ["C"], name="div0")
    graph = helper.make_graph([node], "div_graph", [vi_a, vi_b], [vi_c])
    model = helper.make_model(graph, opset_imports=[helper.make_opsetid("", OPSET)],
                              ir_version=IR_VERSION)
    sess = ort.InferenceSession(model.SerializeToString(),
                                providers=["CPUExecutionProvider"])
    (c,) = sess.run(["C"], {"A": a, "B": b})
    return c


def _bits_equal(want, got) -> bool:
    """Bit-for-bit equality, NaN equal to NaN, of two arrays of the same type."""
    w = np.asarray(want)
    g = np.asarray(got)
    if w.shape != g.shape or w.dtype != g.dtype:
        return False
    if w.dtype.kind == "f":
        uint = UINT_OF_WIDTH[w.dtype.itemsize]
        bw = np.ascontiguousarray(w).view(uint)
        bg = np.ascontiguousarray(g).view(uint)
        both = np.isnan(w) & np.isnan(g)
        return bool(np.all((bw == bg) | both))
    return bool(np.all(w == g))


def _diff_indices(want, got, limit=5):
    """Flat indices where two arrays differ (NaN equal to NaN); None on shape/type."""
    w = np.asarray(want)
    g = np.asarray(got)
    if w.shape != g.shape or w.dtype != g.dtype:
        return None
    if w.dtype.kind == "f":
        uint = UINT_OF_WIDTH[w.dtype.itemsize]
        bw = np.ascontiguousarray(w).view(uint)
        bg = np.ascontiguousarray(g).view(uint)
        both = np.isnan(w) & np.isnan(g)
        mask = ~((bw == bg) | both)
    else:
        mask = ~(w == g)
    idx = [int(i) for i in np.nonzero(mask.ravel())[0]]
    return idx if limit is None else idx[:limit]

def _pow2(e: int) -> Fraction:
    return Fraction(2) ** e


def _round_half_even(x: Fraction) -> int:
    """roundTiesToEven of a non-negative rational to the nearest integer."""
    q = x.numerator // x.denominator
    r = x - q
    half = Fraction(1, 2)
    if r > half:
        return q + 1
    if r < half:
        return q
    return q if q % 2 == 0 else q + 1


def _frac_round_to_type(value: Fraction, dtype):
    """Round an exact rational to the nearest value of `dtype`, ties to even.

    The exponent range is unbounded while deciding the rounding, as the document states:
    a magnitude whose nearest value exceeds the largest finite value of the type gives an
    infinity of the sign of the value, and one below the smallest subnormal gives a zero
    of that sign.
    """
    dt = np.dtype(dtype)
    f = np.finfo(dt)
    p = f.nmant + 1
    emin = int(f.minexp)
    max_finite = Fraction(float(f.max))
    if value == 0:
        return dt.type(0.0)
    sign = -1.0 if value < 0 else 1.0
    a = abs(value)
    e = a.numerator.bit_length() - a.denominator.bit_length()
    while _pow2(e) > a:
        e -= 1
    while _pow2(e + 1) <= a:
        e += 1
    if e < emin:
        e = emin
    spacing = _pow2(e - p + 1)
    n = _round_half_even(a / spacing)
    result = n * spacing
    if result > max_finite:
        return dt.type(sign * float("inf"))
    return dt.type(sign * float(result))


def _exact_float_div(x, y, dtype):
    """C[i] of **Div** (float, float), the document's case analysis, exactly."""
    dt = np.dtype(dtype)
    fx, fy = float(x), float(y)
    if np.isnan(fx) or np.isnan(fy):
        return dt.type(float("nan"))
    sx = bool(np.signbit(fx))
    sy = bool(np.signbit(fy))
    negative = sx != sy
    ainf = bool(np.isinf(fx))
    binf = bool(np.isinf(fy))
    azero = fx == 0.0
    bzero = fy == 0.0
    if (ainf and binf) or (azero and bzero):
        return dt.type(float("nan"))            # the invalid operation of IEEE 7.2
    if (not ainf and not azero and bzero) or (ainf and not binf):
        return dt.type(-np.inf if negative else np.inf)
    if (azero and not bzero) or (not ainf and not azero and binf):
        return dt.type(-0.0 if negative else 0.0)
    return _frac_round_to_type(Fraction(fx) / Fraction(fy), dt)


def _f32_then_f16(x, y):
    """The same quotient computed in float32 and narrowed, or None.

    A float16 kernel that divides in float32 rounds twice; on the rare operand pair
    whose exact quotient sits a hair from a float16 midpoint that differs from the single
    rounding the document states.  It is recorded, not required.
    """
    fx, fy = float(x), float(y)
    if fy == 0.0 or np.isnan(fx) or np.isnan(fy) or np.isinf(fx) or np.isinf(fy):
        return None
    q = np.float32(fx) / np.float32(fy)
    if np.isinf(q):
        return np.float16(q)
    return np.float16(q)

def _doc_int_pair(x: int, y: int, n: int, signed: bool) -> int:
    """trunc(x / y), reduced modulo 2**n when it leaves the range (signed only)."""
    q = abs(x) // abs(y)
    if (x < 0) != (y < 0):
        q = -q
    if signed:
        lo, hi = -(2 ** (n - 1)), 2 ** (n - 1) - 1
        if q > hi or q < lo:
            q -= 2 ** n
    return q


def _doc_int_array(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    dt = a.dtype
    n = dt.itemsize * 8
    signed = dt.kind == "i"
    flat = [_doc_int_pair(int(x), int(y), n, signed)
            for x, y in zip(a.ravel().tolist(), b.ravel().tolist())]
    return np.array(flat, dtype=dt).reshape(a.shape)


def _mk(values, dt):
    """An array of `dt` built from Python ints, without an out-of-range literal.

    A negative literal handed to an unsigned `dtype=` constructor raises under NumPy 2,
    and a value above 2**63-1 does not fit an int64 literal either: the values are built
    in the widest unsigned type and cast, which wraps.
    """
    d = np.dtype(dt)
    vals = [int(v) for v in values]
    if d.kind == "u":
        return np.array(vals, dtype=np.uint64).astype(d)
    return np.array(vals, dtype=np.int64).astype(d)


def _rand_bits(rng, d, n):
    """n random values of `d`, drawn uniformly over the bit patterns of the type."""
    raw = rng.integers(0, 2 ** 64, size=n, dtype=np.uint64)
    bits = d.itemsize * 8
    if bits < 64:
        raw = raw & np.uint64((1 << bits) - 1)
    return raw.astype(np.dtype("u%d" % d.itemsize)).view(d)

def _int_probe(d, rng):
    """The operand pairs of the integer sweep: the 8-bit types exhaustively."""
    info = np.iinfo(d)
    if d.itemsize == 1:
        xs = np.arange(int(info.min), int(info.max) + 1, dtype=np.int64)
        aa = np.repeat(xs, xs.size)
        bb = np.tile(xs, xs.size)
        keep = bb != 0                       # a null divisor is not a case
        aa = aa[keep].astype(d)
        bb = bb[keep].astype(d)
        return aa, bb, "exhaustive (%d pairs)" % aa.size

    lo, hi = int(info.min), int(info.max)
    edges = sorted({lo, lo + 1, lo + 2, hi - 1, hi, 0, 1, 2, 3, 7, 10})
    pairs = [(x, y) for x in edges for y in edges if y != 0]
    if d.kind == "i":
        pairs = [(x, y) for (x, y) in pairs if not (x == lo and y == -1)]

    ra = _rand_bits(rng, d, 200)
    rb = _rand_bits(rng, d, 200)
    rb = np.where(rb == 0, np.asarray(1, dtype=d), rb).astype(d)
    if d.kind == "i":
        keep = ~((ra == info.min) & (rb == d.type(-1)))
        ra, rb = ra[keep], rb[keep]

    aa = np.concatenate([_mk([p[0] for p in pairs], d), ra])
    bb = np.concatenate([_mk([p[1] for p in pairs], d), rb])
    return aa, bb, "%d boundary and %d random pairs" % (len(pairs), ra.size)


def sweep_int_domain(impl):
    """The document's integer case analysis against the implementation and the runtime.

    The two 8-bit types are covered exhaustively over every pair with a non-null
    divisor; the wider types use their boundary values and 200 random pairs.  A null
    divisor is excluded: the document states that the result is undetermined there.
    """
    rng = np.random.default_rng(20261122)
    rows, failures = [], []
    for dt in ALL_INT:
        d = np.dtype(dt)
        try:
            aa, bb, note = _int_probe(d, rng)
        except Exception as exc:
            failures.append({"type": d.name, "who": "the probe",
                             "detail": "%s: %s" % (type(exc).__name__, exc)})
            continue
        want = _doc_int_array(aa, bb)

        try:
            got = np.asarray(impl.div(aa, bb))
        except Exception as exc:
            failures.append({"type": d.name, "who": "the implementation",
                             "detail": "%s: %s" % (type(exc).__name__, exc)})
            got = None
        if got is not None:
            bad = _diff_indices(want, got)
            if bad is None:
                failures.append({"type": d.name,
                                 "who": "the implementation vs the document",
                                 "detail": "shape %s or type %s differs from the "
                                           "document's %s / %s"
                                           % (got.shape, got.dtype, want.shape, want.dtype)})
            elif bad:
                failures.append({
                    "type": d.name, "who": "the implementation vs the document",
                    "detail": "; ".join(
                        "%d / %d: document %d, implementation %d"
                        % (int(aa.ravel()[i]), int(bb.ravel()[i]),
                           int(want.ravel()[i]), int(got.ravel()[i])) for i in bad)})

        try:
            ref = run_ort_batch(aa, bb)
        except Exception as exc:
            rows.append("%s: %s, not reference-checked (%s: %s)"
                        % (d.name, note, type(exc).__name__, exc))
            continue
        bad = _diff_indices(want, ref)
        if bad is None:
            failures.append({"type": d.name, "who": "the document vs ONNX Runtime",
                             "detail": "shape %s or type %s differs from the document's "
                                       "%s / %s" % (ref.shape, ref.dtype,
                                                    want.shape, want.dtype)})
        elif bad:
            failures.append({
                "type": d.name, "who": "the document vs ONNX Runtime",
                "detail": "; ".join(
                    "%d / %d: document %d, ort %d"
                    % (int(aa.ravel()[i]), int(bb.ravel()[i]),
                       int(want.ravel()[i]), int(ref.ravel()[i])) for i in bad)})
        rows.append("%s: %s" % (d.name, note))

    return {"sweep": "int domain (document vs implementation vs ONNX Runtime)",
            "summary": "%d types: %s" % (len(rows), ", ".join(rows)),
            "failures": failures}


def sweep_int_out_of_range(impl):
    """The one quotient outside the range of a signed type: (-2**(n-1)) / -1.

    The document states the result of the second case of its definition: the exact
    quotient 2**(n-1) reduced modulo 2**n, which is -2**(n-1).  The pair is decided here
    by exact arithmetic only.  For int32 and int64 it is never handed to ONNX Runtime,
    whose C division overflows and lets the process die with SIGFPE, and never to the
    implementation either: the lean conformance replay passes ONNX Runtime as `impl`, so
    calling it would kill the driver.  The 8-bit and 16-bit pairs are also covered by the
    exhaustive domain sweep, where the operation is promoted to int and is safe.
    """
    rows, failures = [], []
    for dt in SIGNED_INT:
        d = np.dtype(dt)
        n = d.itemsize * 8
        lo = -(2 ** (n - 1))
        q = _doc_int_pair(lo, -1, n, True)
        rows.append("%s: (%d) / -1 -> %d" % (d.name, lo, q))
        if q != lo:
            failures.append({"type": d.name, "who": "the document's own rule",
                             "detail": "(-2**%d) / -1 must give -2**%d = %d, the "
                                       "document's second case gives %d"
                                       % (n - 1, n - 1, lo, q)})
    return {"sweep": "the quotient outside the range of the signed types "
                     "(exact arithmetic, not sent to the runtime)",
            "summary": "4 signed widths, exact arithmetic only; %s"
                       % ", ".join(rows),
            "failures": failures}
