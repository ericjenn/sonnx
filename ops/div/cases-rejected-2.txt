def run_ort_batch(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """One Div node over two arrays of the same type."""
    proto = helper.np_dtype_to_tensor_dtype(a.dtype)
    vi_a = helper.make_tensor_value_info("A", proto, list(a.shape))
    vi_b = helper.make_tensor_value_info("B", proto, list(b.shape))
    vi_c = helper.make_tensor_value_info("C", proto, None)
    node = helper.make_node(OP, ["A", "B"], ["C"], name="div0")
    graph = helper.make_graph([node], "div_graph", [vi_a, vi_b], [vi_c])
    model = helper.make_model(
        graph,
        opset_imports=[helper.make_opsetid("", OPSET)],
        ir_version=IR_VERSION,
    )
    sess = ort.InferenceSession(
        model.SerializeToString(), providers=["CPUExecutionProvider"]
    )
    (c,) = sess.run(["C"], {"A": a, "B": b})
    return c


def _bits_equal(want: np.ndarray, got: np.ndarray) -> bool:
    if want.shape != got.shape or want.dtype != got.dtype:
        return False
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        return bool(np.all((bw == bg) | both_nan))
    return bool(np.all(want == got))


def _diff_indices(want: np.ndarray, got: np.ndarray, limit=5):
    if want.shape != got.shape or want.dtype != got.dtype:
        return None
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        mask = ~((bw == bg) | both_nan)
    else:
        mask = ~(want == got)
    idx = [int(i) for i in np.nonzero(mask.ravel())[0]]
    return idx[:limit] if limit is not None else idx


def _same_bits(a, b) -> bool:
    aa = np.asarray(a).ravel()
    bb = np.asarray(b).ravel()
    return _bits_equal(aa, bb)

def _pow2(e: int) -> Fraction:
    return Fraction(2) ** e


def _round_half_even(x: Fraction) -> int:
    q = x.numerator // x.denominator
    r = x - q
    half = Fraction(1, 2)
    if r > half:
        return q + 1
    if r < half:
        return q
    return q if q % 2 == 0 else q + 1


def _round_fraction_to_type(value: Fraction, dtype, sign: float = 1.0):
    """Round a positive rational to `dtype`, ties to even, with unbounded exponent."""
    dt = np.dtype(dtype)
    if value == 0:
        return dt.type(sign * 0.0)
    if value < 0:
        value = -value
        sign = -sign
    f = np.finfo(dt)
    p = f.nmant + 1
    emin = f.minexp
    max_finite = Fraction(float(f.max))
    a = value
    e = a.numerator.bit_length() - a.denominator.bit_length()
    while _pow2(e) > a:
        e -= 1
    while _pow2(e + 1) <= a:
        e += 1
    if e < emin:
        e = emin
    spacing = _pow2(e - p + 1)
    n = _round_half_even(a / spacing)
    result = n * spacing
    if result > max_finite:
        return dt.type(sign * float("inf"))
    if n == 0:
        return dt.type(sign * 0.0)
    return dt.type(sign * float(result))


def _exact_float_result(x, y, dtype):
    """The document's float division rule, in exact arithmetic."""
    dt = np.dtype(dtype)
    fx = float(x)
    fy = float(y)
    if np.isnan(fx) or np.isnan(fy):
        return dt.type(float("nan"))
    sx = np.signbit(fx)
    sy = np.signbit(fy)
    sign = -1.0 if sx != sy else 1.0
    if np.isinf(fx) and np.isinf(fy):
        return dt.type(float("nan"))
    if fx == 0.0 and fy == 0.0:
        return dt.type(float("nan"))
    if np.isinf(fx):
        return dt.type(sign * float("inf"))
    if np.isinf(fy):
        return dt.type(sign * 0.0)
    if fy == 0.0:
        return dt.type(sign * float("inf"))
    if fx == 0.0:
        return dt.type(sign * 0.0)
    q = Fraction(fx) / Fraction(fy)
    if q < 0:
        sign = -sign
        q = -q
    return _round_fraction_to_type(q, dt, sign)


def _double_rounded_float16_div(x, y):
    """The exact quotient rounded through float32 and then to float16, or None."""
    fx = float(x)
    fy = float(y)
    if not (np.isfinite(fx) and np.isfinite(fy)) or fy == 0.0:
        return None
    q = Fraction(fx) / Fraction(fy)
    sign = -1.0 if q < 0 else 1.0
    q = abs(q)
    f32 = _round_fraction_to_type(q, np.float32, sign)
    return np.float16(f32)

def _trunc_div(x: int, y: int) -> int:
    q = abs(x) // abs(y)
    return -q if (x < 0) != (y < 0) else q


def _doc_int_value(x: int, y: int, n: int, signed: bool) -> int:
    q = _trunc_div(x, y)
    if signed:
        lo = -(2 ** (n - 1))
        hi = 2 ** (n - 1) - 1
        if q > hi:
            q -= 2 ** n
        elif q < lo:
            q += 2 ** n
    return q


def _doc_int_array(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    dt = a.dtype
    n = dt.itemsize * 8
    signed = dt.kind == "i"
    flat = [_doc_int_value(int(x), int(y), n, signed)
            for x, y in zip(a.ravel(), b.ravel())]
    return np.array(flat, dtype=dt).reshape(a.shape)


def _rand_values(rng, dt, n):
    hi = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    lo = rng.integers(0, 2 ** 32, size=n, dtype=np.uint64)
    raw = (hi << np.uint64(32)) | lo
    bits = dt.itemsize * 8
    if bits < 64:
        raw = raw & np.uint64((1 << bits) - 1)
    return raw.astype(np.dtype("u%d" % dt.itemsize)).view(dt)

def sweep_schema_types(impl):
    declared = [
        "tensor(float16)", "tensor(float)", "tensor(double)",
        "tensor(int8)", "tensor(int16)", "tensor(int32)", "tensor(int64)",
        "tensor(uint8)", "tensor(uint16)", "tensor(uint32)", "tensor(uint64)",
    ]
    try:
        from onnx import defs
        schema = defs.get_schema(OP, OPSET)
        allowed = set()
        for c in schema.type_constraints:
            if c.type_param_str == "T":
                allowed = set(c.allowed_type_strs)
    except Exception as exc:
        return {"sweep": "document types against the ONNX schema",
                "summary": "schema unavailable: %s: %s" % (type(exc).__name__, exc),
                "failures": []}
    failures = []
    for name in declared:
        if name not in allowed:
            failures.append({"who": "the document declares %s" % name,
                             "detail": "the schema of %s-%d admits %s for T"
                                       % (OP, OPSET, sorted(allowed))})
    extra = sorted(allowed - set(declared))
    summary = "%d declared types checked against %s-%d" % (len(declared), OP, OPSET)
    if extra:
        summary += "; the schema also admits %s, not specified by the document" % extra
    return {"sweep": "document types against the ONNX schema",
            "summary": summary, "failures": failures}

def sweep_int_overflow_pair(impl):

def _float_probe(dt, rng):
    dt = np.dtype(dt)
    f = np.finfo(dt)
    small = dt.type(np.nextafter(dt.type(0.0), dt.type(1.0)))
    tiny = dt.type(f.tiny)
    specials = [
        dt.type(0.0), dt.type(-0.0), dt.type(1.0), dt.type(-1.0),
        dt.type(0.5), dt.type(-0.5), dt.type(2.0), dt.type(-2.0),
        dt.type(3.0), dt.type(-3.0), tiny, dt.type(-float(tiny)),
        small, dt.type(-float(small)), dt.type(f.max), dt.type(-f.max),
        dt.type(np.inf), dt.type(-np.inf), dt.type(np.nan),
    ]
    pairs = [(x, y) for x in specials for y in specials]
    n = 80
    ra = (rng.random(n) * 2.0 - 1.0) * float(f.max)
    rb = (rng.random(n) * 2.0 - 1.0) * float(f.max)
    ra[:20] = (rng.random(20) * 2.0 - 1.0) * float(f.tiny)
    rb[:20] = (rng.random(20) * 2.0 - 1.0) * float(f.tiny)
    pairs += list(zip(ra.tolist(), rb.tolist()))
    a = np.array([p[0] for p in pairs], dtype=dt)
    b = np.array([p[1] for p in pairs], dtype=dt)
    return a, b


def _compare_to_exact(who, got, exact, a, b, dt, failures):
    recorded = 0
    bad = _diff_indices(exact, got, limit=None)
    if bad is None:
        failures.append({"type": dt.name, "who": who,
                         "detail": "shape or type differs: exact %s/%s, %s %s/%s"
                                   % (exact.shape, exact.dtype, who, got.shape, got.dtype)})
        return 0
    real_bad = []
    for i in bad:
        if dt == np.dtype(np.float16):
            dr = _double_rounded_float16_div(a.ravel()[i], b.ravel()[i])
            if dr is not None and _same_bits(dr, got.ravel()[i]):
                recorded += 1
                continue
        real_bad.append(i)
    if real_bad:
        failures.append({
            "type": dt.name, "who": who,
            "detail": "; ".join(
                "%r / %r: exact %r, %s %r"
                % (float(a.ravel()[i]), float(b.ravel()[i]),
                   float(exact.ravel()[i]), who, float(got.ravel()[i]))
                for i in real_bad[:5])})
    return recorded
