"""The cases with which the **Atan** specification is checked.

Read by `specloop.py`:

  * ``cases()``   -- one labelled operand per case.  Each case goes through a one-node
                     Atan model (opset 14) and through the implementation, and the two
                     results are compared bit for bit.
  * ``sweeps()``  -- the bulk checks, which build their own ONNX Runtime calls because
                     they must be batched into a single run: the special values of every
                     type, the sign and the shape, the correctness of a result, the
                     domain of the type, the worked examples of the document, and the
                     real section, which has no ONNX type and is checked against exact
                     arithmetic instead.

The document specifies ``round(arctan(x))`` with roundTiesToEven, and a kernel is free to
be less accurate than that.  The reference is therefore not the bit pattern of ONNX
Runtime: the correctness of a result is measured in units of the last place of the value
1, per type, and reported rather than required (see the sweep "the correctness of a
result").  The special values, the sign, the shape and the type are required exactly.

The reference of the bulk sweeps is ``np.arctan`` in float64, rounded to the type: the
document leaves the accuracy of a result to the profile's guidelines, and the departure
is reported, not required.  The exact ``Decimal`` reference is kept for the real section,
where a handful of values is enough; it reduces its argument so that the series converges
in a few dozen terms instead of tens of thousands.

Every case of every section's formula is meant to appear here, and the case set is part
of the loop: when a section is added to the document, its cases are added here before

from __future__ import annotations

from decimal import Decimal, getcontext, localcontext, ROUND_HALF_EVEN

import numpy as np

import onnxruntime as ort
from onnx import helper

OP = "Atan"
OPSET = 14

ALL_FLOAT = [np.float16, np.float32, np.float64]
UINT_OF_WIDTH = {2: np.uint16, 4: np.uint32, 8: np.uint64}
IR_VERSION = 10

# The departure of a result from the correctly rounded value, in units of the last
# place of the value 1, per type.  Four units is not an accuracy difference between two
# conforming implementations but a defect; the departures are reported anyway.
SLACK = 4.0

getcontext().prec = 80


# ---------------------------------------------------------------------------
# ONNX Runtime, batched: one session per element type, reused by every sweep
# ---------------------------------------------------------------------------

_SESSIONS = {}


def _session(dtype):
    """One one-node Atan session per element type, with an unknown shape."""
    dt = np.dtype(dtype)
    key = dt.str
    if key not in _SESSIONS:
        proto = helper.np_dtype_to_tensor_dtype(dt)
        vi_x = helper.make_tensor_value_info("X", proto, None)
        vi_y = helper.make_tensor_value_info("Y", proto, None)
        node = helper.make_node(OP, ["X"], ["Y"], name="atan0")
        graph = helper.make_graph([node], "atan_graph", [vi_x], [vi_y])
        model = helper.make_model(graph, opset_imports=[helper.make_opsetid("", OPSET)],
                                  ir_version=IR_VERSION)
        _SESSIONS[key] = ort.InferenceSession(
            model.SerializeToString(), providers=["CPUExecutionProvider"])
    return _SESSIONS[key]


def run_ort_batch(x: np.ndarray) -> np.ndarray:
    x = np.asarray(x)
    sess = _session(x.dtype)
    (y,) = sess.run(["Y"], {"X": x})
    return y


def _bits_equal(want: np.ndarray, got: np.ndarray) -> bool:
    if want.shape != got.shape or want.dtype != got.dtype:
        return False
    if want.dtype.kind == "f":
        uint = UINT_OF_WIDTH[want.dtype.itemsize]
        bw = np.ascontiguousarray(want).view(uint)
        bg = np.ascontiguousarray(got).view(uint)
        both_nan = np.isnan(want) & np.isnan(got)
        return bool(np.all((bw == bg) | both_nan))
    return bool(np.all(want == got))


# ---------------------------------------------------------------------------
# an independent reference: the arctangent in high-precision Decimal
# ---------------------------------------------------------------------------
#
# The operand of the reference is the value the tensor holds, not the decimal that was
# written to build it: Decimal(v) applied to the float is exact, Decimal(repr(v)) is the
# shortest decimal representation and a different number.  The series is the Taylor
# series of the arctangent, summed in Decimal with a wide context; the argument is
# reduced so that the series converges in a few dozen terms:
#
#   atan(x) = pi/2 - atan(1/x)          for x > 1
#   atan(x) = pi/4 + atan((x-1)/(x+1))  for 1/2 < x <= 1, where |(x-1)/(x+1)| <= 1/3
#
# Without the second reduction the series at x = 1 does not converge at all.


def _atan_series(x: Decimal) -> Decimal:
    """arctan(x) for |x| <= 1/3, by the Taylor series, in Decimal."""
    if x == 0:
        return Decimal(0)
    x2 = x * x
    term = x
    total = x
    n = 1
    eps = Decimal(10) ** (-(getcontext().prec - 10))
    while True:
        term = -term * x2
        add = term / (2 * n + 1)
        total += add
        if abs(add) < eps:
            break
        n += 1
    return total


_PI_CACHE = []


def _pi_decimal() -> Decimal:
    """pi, by the Machin formula, in Decimal, computed once."""
    if not _PI_CACHE:
        with localcontext() as ctx:
            ctx.prec = getcontext().prec + 10
            pi = 16 * _atan_series(Decimal(1) / 5) - 4 * _atan_series(Decimal(1) / 239)
        _PI_CACHE.append(+pi)
    return _PI_CACHE[0]


def _atan_decimal(x: Decimal) -> Decimal:
    """arctan of a Decimal, on the branch (-pi/2, pi/2)."""
    if x.is_nan():
        return x
    if x.is_infinite():
        half = _pi_decimal() / 2
        return half if x > 0 else -half
    if x < 0:
        return -_atan_decimal(-x)
    if x > 1:
        return _pi_decimal() / 2 - _atan_decimal(1 / x)
    if x > Decimal("0.5"):
        return _pi_decimal() / 4 + _atan_series((x - 1) / (x + 1))
    return _atan_series(x)


def _atan_exact(x: float) -> Decimal:
    """The arctangent of the float x, from the float's exact value."""
    return _atan_decimal(Decimal(x))


def _round_to_type(y: Decimal, dtype) -> np.ndarray:
    """Round a Decimal to the nearest value of the type, roundTiesToEven."""
    dt = np.dtype(dtype)
    if y.is_nan():
        return np.array([np.nan], dtype=dt)
    if y.is_infinite():
        return np.array([np.inf if y > 0 else -np.inf], dtype=dt)
    if dt == np.dtype(np.float64):
        return np.array([float(y)], dtype=dt)
    if dt == np.dtype(np.float32):
        return np.array([np.float32(float(y))], dtype=dt)
    # float16: round through the exact rational, ties to even
    f = np.finfo(np.float16)
    if y == 0:
        return np.array([0.0], dtype=dt)
    sign = -1.0 if y < 0 else 1.0
    a = abs(y)
    if a >= Decimal(float(f.max)):
        return np.array([sign * np.inf], dtype=dt)
    e = a.adjusted()
    scale = Decimal(2) ** (10 - e)
    scaled = a * scale
    n = int(scaled.to_integral_value(rounding=ROUND_HALF_EVEN))
    if n >= 2048:
        n //= 2
        e += 1
    val = Decimal(n) * (Decimal(2) ** (e - 10))
    return np.array([sign * float(val)], dtype=dt)

# the reference of the bulk sweeps: np.arctan in float64, rounded to the type
# ---------------------------------------------------------------------------


def _reference_atan(x: np.ndarray, dtype) -> np.ndarray:
    """The arctangent of x, from np.arctan in float64, rounded to the type.

    The document leaves the accuracy of a result to the profile's guidelines, so this
    reference is used to *measure* a departure, not to require a bit pattern.  For
    float16 and float32 the float64 arctangent is far more accurate than the type, so
    the rounding to the type is the correctly rounded value but for a double-rounding
    case of measure zero.
    """
    dt = np.dtype(dtype)
    v = np.arctan(np.asarray(x, dtype=np.float64))
    return np.asarray(v, dtype=dt)


def _ulp_of_one(dtype) -> float:
    """One unit in the last place of the value 1, for this type."""
    f = np.finfo(np.dtype(dtype))
    return float(2.0 ** -f.nmant)


def _departure_ulps(got: np.ndarray, want: np.ndarray, dtype) -> np.ndarray:
    """|got - want| in units of the last place of 1, element-wise, NaN -> 0."""
    ulp = _ulp_of_one(dtype)
    g = np.asarray(got, dtype=np.float64)
    w = np.asarray(want, dtype=np.float64)
    d = np.abs(g - w) / ulp
    d = np.where(np.isnan(g) & np.isnan(w), 0.0, d)
    # a NaN where the reference is finite is a departure, not a match
    d = np.where(np.isnan(d), np.inf, d)
    return d

def _special_values(dtype):
    """The special values of the float section, as documented."""
    dt = np.dtype(dtype)
    half_pi = _round_to_type(_pi_decimal() / 2, dt)[0]
    return [
        ("+0.0 -> +0.0", np.array([0.0], dtype=dt), np.array([0.0], dtype=dt)),
        ("-0.0 -> -0.0", np.array([-0.0], dtype=dt), np.array([-0.0], dtype=dt)),
        ("+inf -> nearest(pi/2)", np.array([np.inf], dtype=dt),
         np.array([half_pi], dtype=dt)),
        ("-inf -> nearest(-pi/2)", np.array([-np.inf], dtype=dt),
         np.array([-half_pi], dtype=dt)),
    ]


def sweep_special_values(impl):
    """The special values of every type, against the document and against ONNX Runtime.

    The document decides these exactly: the sign of a null is preserved, an infinity
    gives the value of the type nearest to +-pi/2, and NaN gives a quiet NaN.  The
    reference is the document's own value, computed in Decimal, not the bit pattern of
    the runtime; the runtime's own value is reported, not required.
    """
    failures = []
    rows = []
    for dtype in ALL_FLOAT:
        dt = np.dtype(dtype)
        x = np.array([0.0, -0.0, np.inf, -np.inf, np.nan], dtype=dt)
        try:
            ref = run_ort_batch(x)
            if _bits_equal(ref, np.asarray(impl.atan(x))):
                note = "ORT agrees"
            else:
                note = "ORT differs (reported, not required)"
        except Exception as exc:
            note = "not reference-checked (%s)" % type(exc).__name__
        for label, xv, want in _special_values(dt):
            got = np.asarray(impl.atan(xv))
            if not _bits_equal(want, got):
                failures.append({
                    "who": "%s: %s" % (dt.name, label),
                    "detail": "document %s, implementation %s"
                              % (want.tolist(), got.tolist()),
                })
        # NaN: any quiet NaN of the type is conforming
        got = np.asarray(impl.atan(np.array([np.nan], dtype=dt)))
        if not (got.shape == (1,) and got.dtype == dt and np.isnan(got[0])):
            failures.append({"who": "%s: NaN" % dt.name,
                             "detail": "implementation %s" % got.tolist()})
        rows.append("%s: special values checked, %s" % (dt.name, note))
    return {"sweep": "the special values of every type (document, exact)",
            "summary": "; ".join(rows), "failures": failures}

def _boundary_values(dtype):
    """The boundary values of the type: zeros, extremes, the subnormal range."""
    f = np.finfo(np.dtype(dtype))
    return [0.0, -0.0, 1.0, -1.0, 0.5, -0.5, 2.0, -2.0,
            float(f.tiny), -float(f.tiny), float(f.max), -float(f.max),
            float(f.smallest_subnormal), -float(f.smallest_subnormal)]


def sweep_correctness(impl):
    """The correctness of a result, in units of the last place of 1, per type.

    The document specifies round(arctan(x)) with roundTiesToEven.  A library is not
    required to be correctly rounded for it, so a departure within SLACK units is not a
    failure; the departures are reported anyway, with the count and the worst departure.
    The reference is np.arctan in float64, rounded to the type; the boundary values and
    the subnormal range are always covered, and the random sample is small because the
    departure is reported, not required.
    """
    failures = []
    rows = []
    rng = np.random.default_rng(20261004)
    for dtype in ALL_FLOAT:
        dt = np.dtype(dtype)
        xs = _boundary_values(dt)
        xs += list(rng.uniform(-8.0, 8.0, 20))
        x = np.array(xs, dtype=dt)
        got = np.asarray(impl.atan(x))
        want = _reference_atan(x, dt)
        d = _departure_ulps(got, want, dt)
        worst = float(np.max(d)) if d.size else 0.0
        n_dep = int(np.sum(d > 0.0))
        if worst > SLACK:
            bad = np.nonzero(d > SLACK)[0][:5]
            failures.append({
                "who": "%s: departure beyond SLACK=%.1f ulp" % (dt.name, SLACK),
                "detail": "; ".join("atan(%r): got %r, reference %r, %.2f ulp"
                                    % (float(x[i]), float(got[i]), float(want[i]),
                                       float(d[i])) for i in bad),
            })
        rows.append("%s: %d/%d depart, worst %.2f ulp" % (dt.name, n_dep, d.size, worst))
    return {"sweep": "the correctness of a result (reported, not required: see V1)",
            "summary": "; ".join(rows), "failures": failures}
